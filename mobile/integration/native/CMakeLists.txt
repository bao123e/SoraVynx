cmake_minimum_required(VERSION 3.22)
project(soravynx_game C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(DOBBY_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
add_subdirectory(${DOBBY_SOURCE_DIR} dobby)
add_library(soravynx_game SHARED api.c backend.c runtime.c)
target_include_directories(soravynx_game PRIVATE ${DOBBY_SOURCE_DIR}/include)
target_compile_definitions(soravynx_game PRIVATE _GNU_SOURCE)
target_compile_options(soravynx_game PRIVATE -Wall -Wextra -Werror=implicit-function-declaration -fvisibility=hidden)
target_link_options(soravynx_game PRIVATE -Wl,-z,max-page-size=16384)
target_compile_definitions(dobby_static PRIVATE DOBBY_LOGGING_DISABLE)
target_link_libraries(soravynx_game PRIVATE dobby_static dl)
